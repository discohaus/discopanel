        // Leans cube toward the cursor
        (() => {
          const tilt = document.querySelector('.mc-tilt');
          if (!tilt || !matchMedia('(hover: hover)').matches) return;
          let tx = 0, ty = 0, cx = 0, cy = 0, raf = null;
          const render = () => {
            cx += (tx - cx) * 0.08;
            cy += (ty - cy) * 0.08;
            tilt.style.transform = `rotateX(${cy}deg) rotateY(${cx}deg)`;
            raf = Math.abs(tx - cx) + Math.abs(ty - cy) > 0.01 ? requestAnimationFrame(render) : null;
          };
          const kick = () => { if (raf === null) raf = requestAnimationFrame(render); };
          addEventListener('mousemove', (e) => {
            tx = (e.clientX / innerWidth - 0.5) * 24;
            ty = (0.5 - e.clientY / innerHeight) * 16;
            kick();
          });
          document.documentElement.addEventListener('mouseleave', () => { tx = 0; ty = 0; kick(); });
